cmake_minimum_required(VERSION 3.25)
project(StandingQuickPocket VERSION 0.1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(ENABLE_SKYRIM_SE ON)
set(ENABLE_SKYRIM_AE ON)
set(ENABLE_SKYRIM_VR OFF)
set(COMMONLIB_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/CommonLibSSE-NG" CACHE PATH "CommonLib source")
set(COMMONLIB_PREBUILT_LIBRARY "" CACHE FILEPATH "Optional matching SE/AE CommonLib library")
if(COMMONLIB_PREBUILT_LIBRARY)
 find_package(spdlog CONFIG REQUIRED)
 find_package(directxtk CONFIG REQUIRED)
 add_library(CommonLibSSE STATIC IMPORTED GLOBAL)
 set_target_properties(CommonLibSSE PROPERTIES IMPORTED_LOCATION "${COMMONLIB_PREBUILT_LIBRARY}")
 target_include_directories(CommonLibSSE INTERFACE "${COMMONLIB_SOURCE_DIR}/include")
 target_compile_definitions(CommonLibSSE INTERFACE ENABLE_SKYRIM_SE=1 ENABLE_SKYRIM_AE=1 HAS_SKYRIM_MULTI_TARGETING=1 SPDLOG_FMT_EXTERNAL SPDLOG_COMPILED_LIB)
 target_link_libraries(CommonLibSSE INTERFACE spdlog::spdlog Microsoft::DirectXTK advapi32 bcrypt d3d11 d3dcompiler dbghelp dxgi ole32 version)
 add_library(CommonLibSSE::CommonLibSSE ALIAS CommonLibSSE)
else()
 set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
 set(COMMONLIB_ENABLE_IPO OFF CACHE BOOL "" FORCE)
 set(SKSE_SUPPORT_PATCH_SAFETY OFF CACHE BOOL "" FORCE)
 add_subdirectory("${COMMONLIB_SOURCE_DIR}" commonlib)
endif()
include("${COMMONLIB_SOURCE_DIR}/cmake/CommonLibSSE.cmake")
add_commonlibsse_plugin(StandingQuickPocket NAME "Standing QuickPocket Compatibility" AUTHOR "Maxmc" VERSION 0.1.1 SOURCES src/Plugin.cpp)
target_include_directories(StandingQuickPocket PRIVATE src)
target_precompile_headers(StandingQuickPocket PRIVATE src/PCH.h)
target_compile_definitions(StandingQuickPocket PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(StandingQuickPocket PRIVATE /W4 /permissive- /Z7 /Zc:preprocessor /utf-8)
target_link_options(StandingQuickPocket PRIVATE /DEBUG /OPT:REF /OPT:ICF)
set_target_properties(StandingQuickPocket PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/package/SKSE/Plugins")
enable_testing()
add_executable(AdapterTests tests/SessionTests.cpp)
target_include_directories(AdapterTests PRIVATE src)
add_test(NAME AdapterPolicy COMMAND AdapterTests)



add_executable(HookTests tests/HookSmoke.cpp)
target_include_directories(HookTests PRIVATE src)
target_compile_definitions(HookTests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
add_test(NAME NativeHookSmoke COMMAND HookTests)
